-- FILAMENTOS (stock de materia prima) create table if not exists filamentos ( id uuid default gen_random_uuid() primary key, color text not null, marca text, material text default 'PLA', gramos_disponibles numeric(10,2) default 0, gramos_minimo numeric(10,2) default 100, precio_por_kg numeric(12,2) default 0, color_hex text default '#888888', activo boolean default true, created_at timestamptz default now() ); -- PEDIDOS create table if not exists pedidos ( id uuid default gen_random_uuid() primary key, numero serial, cliente_id uuid references clientes(id), estado text default 'pendiente', fecha_pedido date default current_date, fecha_entrega date, notas text, total numeric(12,2) default 0, created_at timestamptz default now() ); -- ITEMS DE PEDIDO (con consumo de filamento) create table if not exists pedido_items ( id uuid default gen_random_uuid() primary key, pedido_id uuid references pedidos(id) on delete cascade, descripcion text not null, codigo text, cantidad integer default 1, precio_unitario numeric(12,2) default 0, filamento_id uuid references filamentos(id), gramos_por_unidad numeric(10,2) default 0, subtotal numeric(12,2) default 0 ); -- MOVIMIENTOS DE FILAMENTO (entrada/salida) create table if not exists filamento_movimientos ( id uuid default gen_random_uuid() primary key, filamento_id uuid references filamentos(id), tipo text not null, -- 'entrada' | 'salida' | 'ajuste' gramos numeric(10,2) not null, motivo text, pedido_id uuid references pedidos(id), created_at timestamptz default now() ); -- PROVEEDORES create table if not exists proveedores ( id uuid default gen_random_uuid() primary key, nombre text not null, contacto text, email text, telefono text, notas text, created_at timestamptz default now() ); -- COMPRAS create table if not exists compras ( id uuid default gen_random_uuid() primary key, numero serial, proveedor_id uuid references proveedores(id), descripcion text not null, cantidad numeric(12,2), unidad text, precio_total numeric(12,2) default 0, fecha_compra date default current_date, estado text default 'pendiente', filamento_id uuid references filamentos(id), gramos_recibidos numeric(10,2), notas text, created_at timestamptz default now() ); -- PRESUPUESTOS create table if not exists presupuestos ( id uuid default gen_random_uuid() primary key, numero serial, cliente_id uuid references clientes(id), descripcion text, notas text, validez integer default 15, items jsonb default '[]', subtotal numeric(12,2) default 0, descuento_pct numeric(5,2) default 0, descuento_monto numeric(12,2) default 0, iva_pct numeric(5,2) default 0, iva_monto numeric(12,2) default 0, total numeric(12,2) default 0, estado text default 'borrador', wsp_tel text, created_at timestamptz default now() ); -- CALCULADORA (historial) create table if not exists calculos ( id uuid default gen_random_uuid() primary key, nombre text not null, costo_filamento numeric(12,2) default 0, costo_electricidad numeric(12,2) default 0, costo_fallo numeric(12,2) default 0, otros_costos numeric(12,2) default 0, costo_total numeric(12,2) default 0, margen_pct numeric(5,2) default 35, precio_venta numeric(12,2) default 0, iva_pct numeric(5,2) default 0, precio_con_iva numeric(12,2) default 0, ganancia_unitaria numeric(12,2) default 0, cantidad integer default 1, gramos_usados numeric(10,2) default 0, precio_por_kg numeric(12,2) default 0, created_at timestamptz default now() ); -- GALERÍA create table if not exists galeria ( id uuid default gen_random_uuid() primary key, nombre text not null, descripcion text, categoria text, imagen_url text, created_at timestamptz default now() );
